Write Your First Pipeline

Build a synthetic hello-world CSV-to-Bronze pipeline step by step

1Overview

This tutorial walks you through creating a simple CSV-to-Bronze pipeline using the BasePipeline framework. The pipeline reads a CSV file, validates its schema, and writes it to an Iceberg table in the Bronze layer. After completing this, you will understand the pipeline contract well enough to build production pipelines like TradesBronzePipeline and TradesSilverPipeline.

2Create the Pipeline Class

Create a new file at etl/src/pipelines/bronze/hello_world.py. Every pipeline extends BasePipeline and implements extract(), transform(), and defines a target_schema.

"""Hello World Bronze pipeline -- CSV to Iceberg."""
from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from pyspark.sql import DataFrame

from pyspark.sql.types import StructType, StructField, StringType, DoubleType, TimestampType

from src.pipelines.base import BasePipeline, PipelineConfig, MedallionLayer


class HelloWorldBronzePipeline(BasePipeline):
    """Ingest a hello-world CSV file into the Bronze layer."""

    def __init__(self, spark, csv_path: str = "/data/hello_world.csv"):
        config = PipelineConfig(
            name="hello_world_bronze",
            target_layer=MedallionLayer.BRONZE,
            target_table="hello_world",
            target_schema=StructType([
                StructField("id", StringType(), nullable=False),
                StructField("name", StringType(), nullable=True),
                StructField("value", DoubleType(), nullable=True),
                StructField("created_at", TimestampType(), nullable=True),
            ]),
        )
        super().__init__(spark=spark, config=config)
        self.csv_path = csv_path

    def extract(self) -> DataFrame:
        """Read the source CSV file."""
        return self.spark.read.option("header", "true").csv(self.csv_path)

    def transform(self, df: DataFrame) -> DataFrame:
        """Apply basic cleaning: trim whitespace, cast types."""
        from pyspark.sql.functions import trim, col
        return (
            df
            .withColumn("id", trim(col("id")))
            .withColumn("name", trim(col("name")))
            .withColumn("value", col("value").cast("double"))
            .withColumn("created_at", col("created_at").cast("timestamp"))
        )

3Define the Schema

The target_schema in PipelineConfig defines the expected output. BasePipeline.validate_schema() checks the DataFrame against this schema after transform() runs. If columns are missing or types differ, a SchemaValidationError is raised.

from pyspark.sql.types import StructType, StructField, StringType, DoubleType, TimestampType

schema = StructType([
    StructField("id", StringType(), nullable=False),
    StructField("name", StringType(), nullable=True),
    StructField("value", DoubleType(), nullable=True),
    StructField("created_at", TimestampType(), nullable=True),
])

4Implement extract()

The extract() method reads raw data from the source. For Bronze pipelines, this is typically a file read or API call. The returned DataFrame is the raw, unmodified data.

def extract(self) -> DataFrame:
    """Read the source CSV file with header row."""
    return self.spark.read.option("header", "true").csv(self.csv_path)

5Implement transform()

The transform() method applies cleaning logic. For Bronze, keep it minimal: trim whitespace, cast types, add metadata columns. Heavy business logic belongs in Silver/Gold.

def transform(self, df: DataFrame) -> DataFrame:
    """Trim whitespace and cast columns to target types."""
    from pyspark.sql.functions import trim, col
    return (
        df
        .withColumn("id", trim(col("id")))
        .withColumn("name", trim(col("name")))
        .withColumn("value", col("value").cast("double"))
        .withColumn("created_at", col("created_at").cast("timestamp"))
    )

6Add Quality Checks

BasePipeline supports Soda quality checks via the quality_checks_path config field. Create a Soda YAML file to define data quality rules.

# etl/quality/checks/hello_world_bronze.yml
checks for hello_world:
  - row_count > 0
  - missing_count(id) = 0
  - duplicate_count(id) = 0

Wire Up Quality Checks

Set the quality_checks_path in your PipelineConfig to enable automatic quality gating.

config = PipelineConfig(
    name="hello_world_bronze",
    target_layer=MedallionLayer.BRONZE,
    target_table="hello_world",
    target_schema=schema,
    quality_checks_path="quality/checks/hello_world_bronze.yml",
    critical_checks=["row_count > 0", "missing_count(id) = 0"],
)

7Create an Airflow DAG

Wrap your pipeline in an Airflow DAG for scheduled execution.

# dags/hello_world_dag.py
from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime

from src.pipelines.bronze.hello_world import HelloWorldBronzePipeline
from src.iceberg_utils.catalog import get_spark_session

def run_pipeline():
    spark = get_spark_session("hello_world_bronze")
    pipeline = HelloWorldBronzePipeline(spark)
    result = pipeline.execute()
    print(f"Pipeline result: {result}")
    spark.stop()

with DAG(
    "hello_world_bronze",
    start_date=datetime(2026, 1, 1),
    schedule_interval="@daily",
    catchup=False,
) as dag:
    ingest = PythonOperator(
        task_id="ingest_hello_world",
        python_callable=run_pipeline,
    )

import pytest
from unittest.mock import MagicMock

@pytest.mark.unit
def test_hello_world_pipeline_config():
    """Pipeline config has correct target table and layer."""
    from src.pipelines.bronze.hello_world import HelloWorldBronzePipeline
    from src.pipelines.base import MedallionLayer
    spark = MagicMock()
    pipeline = HelloWorldBronzePipeline(spark)
    assert pipeline.config.target_layer == MedallionLayer.BRONZE
    assert pipeline.config.target_table == "hello_world"
    assert pipeline.config.name == "hello_world_bronze"
